% Normalize macro
\newcommand{\normalize}[3]{%
	\pgfmathparse{(#1 - #2)/(#3 - #2)}%
	\pgfmathresult%
}

% Cell coloring based on normalized value
\newcommand{\colcell}[1]{%
	\pgfmathparse{round(#1*100)}%
	\edef\percent{\pgfmathresult}%
	\cellcolor{red!\percent!green!40}\percent\%
}
